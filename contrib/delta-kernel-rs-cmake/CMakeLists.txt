set(DELTA_KERNEL_RS_BINARY_DIR "${CMAKE_BINARY_DIR}/contrib/delta-kernel-rs")
set(DELTA_KERNEL_RS_CMAKE_DIR "${ClickHouse_SOURCE_DIR}/contrib/delta-kernel-rs-cmake")

include(ExternalProject)

set(USE_DELTA_KERNEL_RS ${ENABLE_LIBRARIES})

# MSAN: Disabled because `ring` is still compiled in, and its hand-written assembly does
#   not generate the symbols required by MSan. Switching to `default-engine-native-tls`
#   only makes `delta-kernel-rs`'s own HTTP client (`reqwest`) able to use ClickHouse's
#   OpenSSL; it does not remove `ring`. `object_store` pulls in `ring` two ways: directly
#   (for AWS request HMAC signing) and transitively through `rustls`, which it forces onto
#   `reqwest` via `rustls-tls-native-roots`. Because Cargo unifies features, `reqwest` ends
#   up built with both the `native-tls` and `rustls` backends, so this is only a partial
#   `native-tls` enablement. Tracking upstream: https://github.com/apache/arrow-rs-object-store/pull/585
if (NOT ENABLE_LIBRARIES OR (NOT OS_LINUX AND NOT OS_DARWIN) OR (NOT ARCH_AMD64 AND NOT ARCH_AARCH64) OR (SANITIZE STREQUAL "memory"))
  message(STATUS "Disabling delta kernel because of incompatible platform or Rust is disabled")
  set(USE_DELTA_KERNEL_RS 0)
endif()

option(ENABLE_DELTA_KERNEL_RS "Enable delta-kernel-rs" ${USE_DELTA_KERNEL_RS})
if (NOT ENABLE_DELTA_KERNEL_RS OR NOT ENABLE_RUST)
  message(STATUS "Not using delta-kernel-rs")
  return()
endif()

clickhouse_import_crate(MANIFEST_PATH "${RUST_CONTRIB_WORKSPACE_MANIFEST}" CRATES _ch_rust_delta_kernel)
clickhouse_config_crate_flags(_ch_rust_delta_kernel)

# Path for autogenerated files (`delta_kernel_ffi.hpp`)
set(DELTA_KERNEL_TARGET_PATH ${DELTA_KERNEL_RS_BINARY_DIR}/include)
# Read only by the build script of `delta_kernel_ffi` (where it writes the headers); cargo gets `--target-dir` from corrosion.
corrosion_set_env_vars(_ch_rust_delta_kernel "CARGO_TARGET_DIR=${DELTA_KERNEL_TARGET_PATH}")

target_include_directories(_ch_rust_delta_kernel INTERFACE "${DELTA_KERNEL_TARGET_PATH}/")
target_link_libraries(_ch_rust_delta_kernel INTERFACE OpenSSL::Crypto OpenSSL::SSL)
add_dependencies(_cargo-build__ch_rust_delta_kernel OpenSSL::Crypto OpenSSL::SSL)

# macOS frameworks required by reqwest's system-configuration dependency
if (OS_DARWIN)
    target_link_libraries(_ch_rust_delta_kernel INTERFACE "-framework SystemConfiguration")
    target_link_libraries(_ch_rust_delta_kernel INTERFACE "-framework CoreFoundation")
    target_link_libraries(_ch_rust_delta_kernel INTERFACE "-framework Security")
endif()

add_library(ch_rust::delta_kernel_rs ALIAS _ch_rust_delta_kernel)
