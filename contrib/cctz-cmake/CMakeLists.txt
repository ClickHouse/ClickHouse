set(LIBRARY_DIR "${ClickHouse_SOURCE_DIR}/contrib/cctz")

set (SRCS
    "${LIBRARY_DIR}/src/civil_time_detail.cc"
    "${LIBRARY_DIR}/src/time_zone_fixed.cc"
    "${LIBRARY_DIR}/src/time_zone_format.cc"
    "${LIBRARY_DIR}/src/time_zone_if.cc"
    "${LIBRARY_DIR}/src/time_zone_impl.cc"
    "${LIBRARY_DIR}/src/time_zone_info.cc"
    "${LIBRARY_DIR}/src/time_zone_libc.cc"
    "${LIBRARY_DIR}/src/time_zone_lookup.cc"
    "${LIBRARY_DIR}/src/time_zone_posix.cc"
    "${LIBRARY_DIR}/src/zone_info_source.cc"
)

add_library (_cctz ${SRCS})
target_include_directories (_cctz PUBLIC "${LIBRARY_DIR}/include")

if (OS_FREEBSD)
    # yes, need linux, because bsd check inside linux in time_zone_libc.cc:24
    target_compile_definitions (_cctz PRIVATE __USE_BSD linux _XOPEN_SOURCE=600)
endif ()

# Related to time_zones table:
# TimeZones.generated.cpp is rewritten only when its content changes; the time zone files are its dependencies via #embed.
set(TIMEZONES_FILE "${CMAKE_CURRENT_BINARY_DIR}/TimeZones.generated.cpp")
set(TIMEZONES_CONTENT "")

# get the list of timezones from tzdata shipped with cctz
set(TZDIR "${LIBRARY_DIR}/testdata/zoneinfo")
file(STRINGS "${LIBRARY_DIR}/testdata/version" TZDATA_VERSION)
set_property(GLOBAL PROPERTY TZDATA_VERSION_PROP "${TZDATA_VERSION}")
message(STATUS "Packaging with tzdata version: ${TZDATA_VERSION}")

# each file in that dir (except of tab and localtime) store the info about timezone
execute_process(COMMAND
    bash -c "cd ${TZDIR} && find * -type f -and ! -name '*.tab' -and ! -name 'localtime' | LC_ALL=C sort | paste -sd ';' -"
    OUTPUT_STRIP_TRAILING_WHITESPACE
    OUTPUT_VARIABLE TIMEZONES
    COMMAND_ERROR_IS_FATAL ANY
)

string(APPEND TIMEZONES_CONTENT "// autogenerated by ClickHouse/contrib/cctz-cmake/CMakeLists.txt\n")

set (COUNTER 1)
foreach(TIMEZONE ${TIMEZONES})
    string(APPEND TIMEZONES_CONTENT "constexpr unsigned char resource_timezone${COUNTER}[] = {\n#embed \"${TZDIR}/${TIMEZONE}\"\n};\n")
    MATH(EXPR COUNTER "${COUNTER}+1")
endforeach(TIMEZONE)

string(APPEND TIMEZONES_CONTENT "const char * auto_time_zones[] {\n" )

foreach(TIMEZONE ${TIMEZONES})
    string(APPEND TIMEZONES_CONTENT "    \"${TIMEZONE}\",\n")
    MATH(EXPR COUNTER "${COUNTER}+1")
endforeach(TIMEZONE)

string(APPEND TIMEZONES_CONTENT "    nullptr\n};\n\n")

string(APPEND TIMEZONES_CONTENT "#include <string_view>\n\n")
# Takes a `std::string_view`, not a `const char *`: a name is also the key of the `DateLUT` cache and
# can come from a length-prefixed carrier (binary type decoding, a SQL literal), so a name with an
# embedded `\0` must not silently match the built-in zone that its prefix names.
string(APPEND TIMEZONES_CONTENT "std::string_view getTimeZone(std::string_view name)\n{\n" )

set (COUNTER 1)
foreach(TIMEZONE ${TIMEZONES})
    string(APPEND TIMEZONES_CONTENT "    if (std::string_view(\"${TIMEZONE}\") == name) return { reinterpret_cast<const char *>(resource_timezone${COUNTER}), std::size(resource_timezone${COUNTER}) };\n")
    MATH(EXPR COUNTER "${COUNTER}+1")
endforeach(TIMEZONE)

string(APPEND TIMEZONES_CONTENT "    return {};\n")
string(APPEND TIMEZONES_CONTENT "}\n")

write_file_if_changed(${TIMEZONES_FILE} "${TIMEZONES_CONTENT}")

add_library (tzdata ${TIMEZONES_FILE})
target_link_libraries(_cctz tzdata)

add_library(ch_contrib::cctz ALIAS _cctz)
