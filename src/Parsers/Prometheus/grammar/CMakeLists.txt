set(ENABLE_ANTLR4_GRAMMARS_DEFAULT ${ENABLE_LIBRARIES})
option(ENABLE_ANTLR4_GRAMMARS "Use ANTLR4 grammars" ${ENABLE_ANTLR4_GRAMMARS_DEFAULT})

if(NOT ENABLE_ANTLR4_GRAMMARS)
  message(STATUS "Not using ANTLR4 grammars")
  return()
endif()

set(_ANTLR4_GRAMMARS_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/generated")
set(_ANTLR4_GRAMMARS_SOURCE_DIR "${_ANTLR4_GRAMMARS_INCLUDE_DIR}/antlr4_grammars")

file(GLOB _ANTLR4_GRAMMARS_SRC "${_ANTLR4_GRAMMARS_SOURCE_DIR}/*.h" "${_ANTLR4_GRAMMARS_SOURCE_DIR}/*.cpp")

add_library(clickhouse_antlr4_grammars ${_ANTLR4_GRAMMARS_SRC})

target_include_directories(clickhouse_antlr4_grammars
    PUBLIC "${_ANTLR4_GRAMMARS_INCLUDE_DIR}"
)

target_link_libraries(clickhouse_antlr4_grammars PUBLIC ch_contrib::antlr4_cpp_runtime)

# The sources are generated by ANTLR, so build them like contrib code:
# without warnings, clang-tidy, coverage instrumentation, UBSan and XRay.
target_compile_options(clickhouse_antlr4_grammars PRIVATE -w ${WITHOUT_COVERAGE_FLAGS_LIST})
set_target_properties(clickhouse_antlr4_grammars PROPERTIES CXX_CLANG_TIDY "")
if (SANITIZE STREQUAL "undefined" OR SANITIZE STREQUAL "address,undefined")
    target_compile_options(clickhouse_antlr4_grammars PRIVATE -fno-sanitize=undefined)
endif()
if (USE_XRAY)
    target_compile_options(clickhouse_antlr4_grammars PRIVATE -fno-xray-instrument)
endif()
