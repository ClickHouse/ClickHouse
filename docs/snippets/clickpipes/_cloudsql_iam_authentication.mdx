export const CloudSQLIAMAuthentication = ({ engine }) => {
  const engines = {
    postgres: {
      docsPath: 'postgres',
      flag: 'cloudsql.iam_authentication',
      port: '5432',
      userName: '<name>@<project>.iam',
      userNameHint: <>For Postgres, the user name is the service account email <strong>without</strong> the <code>.gserviceaccount.com</code> suffix:</>,
      grantsIntro: <>Connect through the admin user and grant the new user the same privileges as the password-based user above. Create the publication as described in the previous section.</>,
      grantsSql: `GRANT USAGE ON SCHEMA "public" TO "<name>@<project>.iam";
GRANT SELECT ON ALL TABLES IN SCHEMA "public" TO "<name>@<project>.iam";
ALTER DEFAULT PRIVILEGES IN SCHEMA "public" GRANT SELECT ON TABLES TO "<name>@<project>.iam";
ALTER USER "<name>@<project>.iam" WITH REPLICATION;`,
      firewallAnchor: '#add-clickpipes-ips-to-firewall',
    },
    mysql: {
      docsPath: 'mysql',
      flag: 'cloudsql_iam_authentication',
      port: '3306',
      userName: '<name>',
      userNameHint: <>For MySQL, the user name is the part of the service account email <strong>before</strong> the <code>@</code>:</>,
      grantsIntro: <>Connect as the root user and grant the new user the same schema and replication privileges as the password-based user above. Repeat the first command for each database you want to replicate:</>,
      grantsSql: `GRANT SELECT ON \`clickpipes\`.* TO '<name>'@'%';
GRANT REPLICATION CLIENT ON *.* TO '<name>'@'%';
GRANT REPLICATION SLAVE ON *.* TO '<name>'@'%';`,
      firewallAnchor: '#configure-network-access-gcp-mysql',
      tlsSuffix: <> You can skip the <a href="#download-root-ca-certificate-gcp-mysql">certificate step</a>.</>,
    },
  };
  const cfg = engines[String(engine).toLowerCase()];

  if (!cfg)
    throw new Error(`Unsupported Cloud SQL IAM authentication engine: ${engine}`);

  const addBindingSql = (role) => `gcloud projects add-iam-policy-binding YOUR_PROJECT_ID \\
  --member="serviceAccount:SERVICE_ACCOUNT_EMAIL" \\
  --role="${role}"`;

  const createUserSql = `gcloud sql users create "${cfg.userName}" \\
  --instance=YOUR_INSTANCE \\
  --type=cloud_iam_service_account`;

  return (
    <>
      <p>
        Instead of a password, ClickPipes can authenticate with the Google Cloud service account that belongs to your ClickHouse Cloud service, using <a href={`https://docs.cloud.google.com/sql/docs/${cfg.docsPath}/iam-authentication#auto-iam-auth`}>automatic IAM database authentication</a>. ClickPipes then connects to your Cloud SQL instance without storing any database credentials.
      </p>

      <Note>
        IAM database authentication is in <strong>Private Preview</strong>. Reach out to the <a href="https://clickhouse.com/support/program">ClickHouse support team</a> to enable it for your service.
      </Note>

      <h3 id="get-the-clickhouse-service-account">Get the ClickHouse service account</h3>

      <p>
        Use the <a href="/products/cloud/api-reference/clickpipes/click-pipes-service-context-get">Get ClickPipes service context</a> API endpoint to get the GCP service account of your ClickHouse service. Its email looks like <code>{'<name>@<project>.iam.gserviceaccount.com'}</code>. The steps below refer to it as <code>SERVICE_ACCOUNT_EMAIL</code>.
      </p>

      <h3 id="configure-cloud-sql-for-iam">Configure Cloud SQL</h3>

      <ol>
        <li>
          <a href="https://console.cloud.google.com/apis/library/sqladmin.googleapis.com">Enable the Cloud SQL Admin API</a> in the project that hosts your instance. ClickPipes uses it to look up the instance and to get a short-lived client certificate.
        </li>
        <li>
          Turn on the <code>{cfg.flag}</code> flag on your instance. Changing this flag restarts the instance.
        </li>
        <li>
          Grant the service account the <code>roles/cloudsql.client</code> and <code>roles/cloudsql.instanceUser</code> IAM roles on the project that hosts your instance:

          <CodeBlock language="bash">{`${addBindingSql('roles/cloudsql.client')}

${addBindingSql('roles/cloudsql.instanceUser')}`}</CodeBlock>
        </li>
        <li>
          Create a database user for the service account. {cfg.userNameHint}

          <CodeBlock language="bash">{createUserSql}</CodeBlock>
        </li>
        <li>
          {cfg.grantsIntro}

          <CodeBlock language="sql">{cfg.grantsSql}</CodeBlock>
        </li>
      </ol>

      <h3 id="create-the-clickpipe-with-iam">Create the ClickPipe</h3>

      <p>
        When you create the ClickPipe, select Google Cloud service account (IAM) authentication. You don't need to enter a username or a password.
      </p>

      <ul>
        <li>Set <strong>Host</strong> to the instance connection name, in the form <code>PROJECT:REGION:INSTANCE</code>. An IP address is not accepted.</li>
        <li>ClickPipes connects to the instance's server-side proxy port, <strong>3307</strong>, and not to port {cfg.port}. Make sure this port is reachable. For a public IP, <a href={cfg.firewallAnchor}>add the ClickPipes IPs to your authorized networks</a>.</li>
        <li>ClickPipes uses Private Service Connect when your service has a <a href="/integrations/clickpipes/networking/gcp-psc">reverse private endpoint</a> that resolves the instance's PSC DNS name. Otherwise, it connects to the instance's public IP, or to its private IP if the instance has no public one.</li>
        <li>TLS settings such as the root CA don't apply. The connection is encrypted and the server identity is verified automatically.{cfg.tlsSuffix}</li>
      </ul>
    </>
  );
};
